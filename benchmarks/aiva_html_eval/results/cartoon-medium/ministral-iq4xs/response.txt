        // Add chat bubble
        function addChatBubble(svg, x, y, direction, content, bubbleClass) {
            const bubbleGroup = document.createElementNS("http://www.w3.org/2000/svg", "g");
            bubbleGroup.setAttribute("transform", `translate(${x},${y})`);

                bubble.setAttribute("points", "0,0 10,0 10,15 0,15 0,0");

            bubble.setAttribute("fill", "#ecf0f1");
            bubble.setAttribute("stroke", "#95a5a6");
            bubble.setAttribute("stroke-width", "1");

            const text = document.createElementNS("http://www.w3.org/2000/svg", "text");

            text.setAttribute("class", "chat-text");

            bubbleGroup.appendChild(bubble);
            bubbleGroup.appendChild(text);
            svg.appendChild(bubbleGroup);

            // Add message arrow if needed
            if (direction === "left") {
                const arrow = document.createElementNS("http://www.w3.org/2000/svg", "polygon");
                arrow.setAttribute("points", "10,0 10,15 0,7.5");
                arrow.setAttribute("fill", "#5f4b94");
                bubbleGroup.appendChild(arrow);
            } else if (direction === "right") {
                const arrow = document.createElementNS("http://www.w3.org/2000/svg", "polygon");
                arrow.setAttribute("points", "0,0 0,15 10,7.5");
                arrow.setAttribute("fill", "#5f4b94");
                bubbleGroup.appendChild(arrow);
            }

            return bubbleGroup;
        }

        // Setup step indicator
        function setupStepIndicator() {
            // Clear existing steps
            elements.stepIndicator.innerHTML = '';

            // Create step indicators
            for (let i = 0; i < demo.steps; i++) {
                const stepDiv = document.createElement('div');
                stepDiv.className = 'step';
                stepDiv.dataset.index = i;

                const stepNumber = document.createElement('div');
                stepNumber.className = 'step-number';
                stepNumber.textContent = (i + 1).toString();

                const stepLabel = document.createElement('div');
                stepLabel.className = 'step-label';
                stepLabel.textContent = demo.stepDescriptions[i];

                stepDiv.appendChild(stepNumber);
                stepDiv.appendChild(stepLabel);
                elements.stepIndicator.appendChild(stepDiv);
            }

            // Add click handlers for step navigation
            document.querySelectorAll('.step').forEach(step => {
                step.addEventListener('click', () => {
                    if (!demo.isPlaying) {
                        const index = parseInt(step.dataset.index);
                        demo.currentStep = index;
                        updateDemo();
                    }
                });
            });
        }

        // Update demo UI
        function updateDemo() {
            // Update step label
            elements.stepLabel.textContent = demo.stepDescriptions[demo.currentStep];
            elements.stepLabel.className = 'hidden';
            if (demo.currentStep >= 0 && demo.currentStep < demo.steps) {
                elements.stepLabel.className = 'step-label';
            }

            // Update step indicator highlights
            document.querySelectorAll('.step').forEach((el, index) => {
                el.classList.toggle('active', index === demo.currentStep);
            });

            // Update diagram based on current step
            updateStepVisualization(demo.currentStep);

            // Update summary section if needed
            if (demo.currentStep === demo.steps - 1) {
                updateSummary();
            } else {
                elements.summarySection.classList.add('hidden');
            }
        }

        // Update visualization for current step
        function updateStepVisualization(step) {
            const svg = elements.diagram.querySelector('svg');
            if (!svg) return;

            // Clear existing content except chat interface
            const chatInterface = elements.diagram.querySelector('.chat-interface');
            svg.innerHTML = '';

            // Recreate chat interface
            const chatInterfaceClone = chatInterface.cloneNode(true);
            svg.appendChild(chatInterfaceClone);

            // Only proceed if we have paths defined
            if (!demo.paths[step]) return;

            // Create paths for this step
            const pathData = demo.paths[step];

            // Add message tokens along the path
            const tokensGroup = document.createElementNS("http://www.w3.org/2000/svg", "g");
            tokensGroup.setAttribute("class", "message-tokens");

            // Draw tokens along the path
            const startX = pathData.start.x;
            const startY = pathData.start.y;
            const endX = pathData.end.x;
            const endY = pathData.end.y;

            // Calculate intermediate points for path
            const points = calculatePathPoints(startX, startY, endX, endY);

            // Add tokens along the path
            for (let i = 0; i < pathData.tokens.length; i++) {
                const token = pathData.tokens[i];
                const pos = points[Math.min(i + 1, points.length - 1)];

                const tokenGroup = document.createElementNS("http://www.w3.org/2000/svg", "g");
                tokenGroup.setAttribute("transform", `translate(${pos.x - 10},${pos.y - 10})`);

                const tokenCircle = document.createElementNS("http://www.w3.org/2000/svg", "circle");
                tokenCircle.setAttribute("r", "10");
                tokenCircle.setAttribute("fill", token.color);
                tokenCircle.setAttribute("stroke", "white");
                tokenCircle.setAttribute("stroke-width", "1");

                const tokenText = document.createElementNS("http://www.w3.org/2000/svg", "text");
                tokenText.setAttribute("x", "0");
                tokenText.setAttribute("y", "0");
                tokenText.setAttribute("text-anchor", "middle");
                tokenText.setAttribute("font-size", "8");
                tokenText.setAttribute("fill", "white");
                tokenText.textContent = token.text;

                tokenGroup.appendChild(tokenCircle);
                tokenGroup.appendChild(tokenText);

                // Position tokens along the path with some spacing
                const spacing = pathData.tokens.length > 1 ? 30 : 0;
                const tokenPosX = pos.x;
                const tokenPosY = pos.y;

                // Adjust position for visual appeal
                tokenGroup.setAttribute("transform", `translate(${tokenPosX},${tokenPosY})`);

                tokensGroup.appendChild(tokenGroup);
            }

            svg.appendChild(tokensGroup);

            // Add path line
            const pathLine = document.createElementNS("http://www.w3.org/2000/svg", "path");
            pathLine.setAttribute("d", `M${startX},${startY} ${endX},${endY}`);
            pathLine.setAttribute("stroke", pathData.pathType === 'response' ? '#5f4b94' : '#ecf0f1');
            pathLine.setAttribute("stroke-width", "1");
            pathLine.setAttribute("fill", "none");

            // Add arrowhead
            const arrow = document.createElementNS("http://www.w3.org/2000/svg", "path");
            const arrowPoints = `
                M${endX - 10},${endY} ${endX + 10},${endY + 5} ${endX - 10},${endY + 10} Z
            `;

            if (pathData.pathType === 'response') {
                arrow.setAttribute("d", arrowPoints);
                arrow.setAttribute("fill", "#5f4b94");
            } else {
                arrow.setAttribute("d", arrowPoints);
                arrow.setAttribute("fill", "#ecf0f1");
            }

            svg.appendChild(pathLine);
            svg.appendChild(arrow);

            // Create characters based on step
            for (let i = 0; i < demo.characters.length; i++) {
                const charData = demo.characters[i];
                if (i === 0 || i === demo.currentStep || (i > 0 && demo.currentStep > 0 && i <= demo.currentStep)) {
                    const character = createCharacter(charData, `user-character-${i}`, demo.characters[i].name);

                    // Position character
                    character.setAttribute("transform", `translate(${charData.pos.x - charData.size/2},${charData.pos.y - charData.size/2})`);

                    if (charData.bubble && demo.currentStep >= i) {
                        addChatBubble(
                            svg,
                            charData.bubble.pos.x,
                            charData.bubble.pos.y,
                            charData.bubble.direction,
                            charData.bubble.content,
                            "ai-bubble"
                        );
                    }

                    svg.appendChild(character);
                }
            }
        }

        // Calculate path points for token placement
        function calculatePathPoints(startX, startY, endX, endY) {
            // Simple linear path for demo purposes
            const points = [];

            // Add start point
            points.push({x: startX, y: startY});

            // Add intermediate points for token placement
            const segments = 4;
            for (let i = 1; i <= segments; i++) {
                const ratio = i / segments;
                const x = startX + (endX - startX) * ratio;
                const y = startY + (endY - startY) * ratio;
                points.push({x: x, y: y});
            }

            // Add end point
            points.push({x: endX, y: endY});

            return points;
        }

        // Update summary section

            elements.summarySection.classList.remove('hidden');

            // Clear existing journey steps
            elements.journeySummary.innerHTML = '';

            // Create journey steps
            for (let i = 0; i < demo.journeyIcons.length; i++) {
                const journeyStep = document.createElement('div');
                journeyStep.className = 'journey-step';

                const icon = document.createElement('div');
                icon.className = `journey-icon ${i < demo.currentStep ? 'completed' : ''}`;
                icon.innerHTML = demo.journeyIcons[i].emoji;

                const text = document.createElement('div');
                text.className = `journey-text ${i < demo.currentStep ? 'completed' : ''}`;
                text.textContent = demo.journeyIcons[i].name.toUpperCase();

                journeyStep.appendChild(icon);
                journeyStep.appendChild(text);
                elements.journeySummary.appendChild(journeyStep);
            }
        }

        // Setup event listeners
        function setupEventListeners() {
            // Next button
            elements.nextButton.addEventListener('click', () => {
                if (!demo.isPlaying) {
                    advanceStep();
                }
            });

            // Reset button
            elements.resetButton.addEventListener('click', resetDemo);

            // Play/Pause button
            elements.playButton.addEventListener('click', () => {
                if (demo.isPlaying) {
                    demo.isPlaying = false;
                    demo.isPausedByUser = true;
                    elements.playButton.textContent = "Play Animation";
                } else {
                    if (demo.currentStep < demo.steps) {
                        demo.isPlaying = true;
                        elements.playButton.textContent = "Pause Animation";
                        playAnimation();
                    }
                }
            });

            // Keyboard navigation
            document.addEventListener('keydown', (e) => {
                if (e.key === 'ArrowRight' && !demo.isPlaying) {
                    advanceStep();
                }
                if (e.key === 'ArrowLeft' && !demo.isPlaying) {
                    if (demo.currentStep > 0) {
                        demo.currentStep--;
                        updateDemo();
                    }
                }
                if (e.key === ' ') {
                    if (demo.isPlaying) {
                        demo.isPlaying = false;
                        demo.isPausedByUser = true;
                        elements.playButton.textContent = "Play Animation";
                    } else {
                        if (demo.currentStep < demo.steps) {
                            demo.isPlaying = true;
                            elements.playButton.textContent = "Pause Animation";
                            playAnimation();
                        }
                    }
                }
            });
        }

        // Advance to next step
        function advanceStep() {
            if (demo.currentStep < demo.steps - 1) {
                demo.currentStep++;
                updateDemo();
            }
        }

        // Reset demo
        function resetDemo() {
            demo.currentStep = 0;
            demo.isPlaying = false;
            demo.isPausedByUser = false;
            elements.playButton.textContent = "Play Animation";
            elements.nextButton.disabled = false;
            updateDemo();
        }

        // Play animation
        function playAnimation() {
            if (demo.currentStep >= demo.steps) return;

            if (demo.isPlaying && !demo.isPausedByUser) {
                demo.isPausedByUser = true;
                elements.playButton.textContent = "Play Animation";
                return;
            }

            if (demo.currentStep < demo.steps) {
                demo.isPlaying = true;
                elements.playButton.textContent = "Pause Animation";

                const playInterval = setInterval(() => {
                    if (demo.isPausedByUser) {
                        clearInterval(playInterval);
                        return;
                    }

                    if (demo.currentStep < demo.steps - 1) {
                        demo.currentStep++;
                        updateDemo();
                    } else {
                        demo.isPlaying = false;
                        demo.isPausedByUser = false;
                        elements.playButton.textContent = "Play Animation";
                        clearInterval(playInterval);
                    }
                }, demo.animationSpeed);
            }
        }

        // Toggle playback
        function togglePlayback() {
            if (demo.isPlaying) {
                // Pause
                demo.isPausedByUser = true;
                demo.isPlaying = false;
                elements.playButton.textContent = "Play Animation";
            } else {
                if (demo.currentStep < demo.steps) {
                    // Play
                    demo.isPlaying = true;
                    demo.isPausedByUser = false;
                    elements.playButton.textContent = "Pause Animation";
                    playAnimation();
                }
            }
        }

        // Get demo state
        function getDemoState() {
            return {
                step: demo.currentStep,
                playing: demo.isPlaying
            };
        }

        // Initialize when DOM is ready
        document.addEventListener('DOMContentLoaded', initDemo);
    </script>
